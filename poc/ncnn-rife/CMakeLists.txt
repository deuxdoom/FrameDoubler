cmake_minimum_required(VERSION 3.20)
project(framedoubler_ncnn_poc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT NCNN_SOURCE_DIR)
    message(FATAL_ERROR "Set NCNN_SOURCE_DIR to the pinned, external ncnn source tree")
endif()
add_subdirectory("${NCNN_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ncnn")
add_executable(rife-poc main.cpp warp.cpp hardware.cpp)
target_link_libraries(rife-poc PRIVATE ncnn advapi32)
if(MINGW)
    target_link_options(rife-poc PRIVATE -static -municode)
endif()
